blob转pem借用ph:

```Python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-

from pathlib import Path
import sys
from cryptography.hazmat.primitives.asymmetric import rsa
from cryptography.hazmat.primitives import serialization


def blob_to_pem(input_path: str, output_path: str) -> None:
    data = Path(input_path).read_bytes()

    magic = data[0:4]
    if magic != b'RSA3':
        raise ValueError(f'Unsupported blob magic: {magic!r}, expected b\'RSA3\'')

    bitlen = int.from_bytes(data[4:8], 'little')
    cb_exp = int.from_bytes(data[8:12], 'little')
    cb_modulus = int.from_bytes(data[12:16], 'little')
    cb_prime1 = int.from_bytes(data[16:20], 'little')
    cb_prime2 = int.from_bytes(data[20:24], 'little')

    idx = 24

    def take(n: int) -> bytes:
        nonlocal idx
        chunk = data[idx:idx + n]
        if len(chunk) != n:
            raise ValueError(f'Truncated blob at offset {idx}, need {n} bytes')
        idx += n
        return chunk

    exp = int.from_bytes(take(cb_exp), 'big')
    modulus = int.from_bytes(take(cb_modulus), 'big')
    p = int.from_bytes(take(cb_prime1), 'big')
    q = int.from_bytes(take(cb_prime2), 'big')
    dp = int.from_bytes(take(cb_prime1), 'big')
    dq = int.from_bytes(take(cb_prime2), 'big')
    qi = int.from_bytes(take(cb_prime1), 'big')
    d = int.from_bytes(take(cb_modulus), 'big')

    key = rsa.RSAPrivateNumbers(
        p=p,
        q=q,
        d=d,
        dmp1=dp,
        dmq1=dq,
        iqmp=qi,
        public_numbers=rsa.RSAPublicNumbers(e=exp, n=modulus),
    ).private_key()

    pem = key.private_bytes(
        encoding=serialization.Encoding.PEM,
        format=serialization.PrivateFormat.TraditionalOpenSSL,
        encryption_algorithm=serialization.NoEncryption(),
    )

    Path(output_path).write_bytes(pem)
    print(f'[+] Converted {input_path} -> {output_path}')
    print(f'[+] bitlen={bitlen}, e={exp}')


def main() -> None:
    if len(sys.argv) == 1:
        input_path = 'private_key_dec.txt'
        output_path = 'private_key_dec.pem'
    elif len(sys.argv) == 3:
        input_path = sys.argv[1]
        output_path = sys.argv[2]
    else:
        print(f'Usage: {sys.argv[0]} [input_blob output_pem]')
        print('Default: private.txt -> private.pem')
        sys.exit(1)

    blob_to_pem(input_path, output_path)


if __name__ == '__main__':
    main()
```

```Python
def rot16_re(text):
    result = []
    for ch in text:
        if 'a' <= ch <= 'z':
            result.append(chr((ord(ch) - 97 + (26 - 16) ) % 26 + 97))
        elif 'A' <= ch <= 'Z':
            result.append(chr((ord(ch) - 65 + (26 - 16) ) % 26 + 65))
        else:
            result.append(ch)
    return ''.join(result)

def Decrypt(data,key):
    result = bytearray(len(data))
    key_bytes = key.encode('utf-8')
    key_len = len(key)
    for i in range(len(data)):
        key_char = key_bytes[i % key_len]
        # 核心算法：与 Native 层完全一致
        result[i] = (key_char + (i % key_len)) ^ data[i]

    return result
```
